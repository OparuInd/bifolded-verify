import { createClient } from "npm:@supabase/supabase-js@2";

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "content-type": "application/json",
};

function out(body: unknown, status = 200) {
  return new Response(JSON.stringify(body), { status, headers: CORS });
}

/* The post-contract rail. Merged from the field-verification form's five
   tasks and Shop Ops' two-phase gate model. Steps are DATA — rename, add,
   or n/a them per job; this list only seeds new jobs. */
const STEP_RAIL = [
  { key: "deposit",   phase: "pre",   name: "Deposit received",                      actor: "customer" },
  { key: "verify",    phase: "pre",   name: "Field verification",                    actor: "contractor" },
  { key: "samples",   phase: "pre",   name: "Color & stain samples",                 actor: "tr" },
  { key: "hardware",  phase: "pre",   name: "Custom hardware & upgrades confirmed",  actor: "customer" },
  { key: "drawings",  phase: "pre",   name: "Shop drawings",                         actor: "tr" },
  { key: "signoff",   phase: "gate",  name: "Drawing sign-off — production gate",    actor: "customer" },
  { key: "fab",       phase: "prod",  name: "Fabrication",                           actor: "tr" },
  { key: "walkaround", phase: "prod", name: "Walkaround video — shop sign-off before paint", actor: "tr", hide: true },
  { key: "finish",    phase: "prod",  name: "Finishing",                             actor: "tr" },
  { key: "qc",        phase: "ship",  name: "QC & pre-ship checklist — ship gate",   actor: "tr" },
  { key: "delivery",  phase: "close", name: "Delivery scheduled",                    actor: "customer" },
  { key: "delivered", phase: "close", name: "Delivered",                             actor: "tr" },
  { key: "install_media", phase: "close", name: "Install photos & video",             actor: "tr", hide: true },
  { key: "balance",   phase: "close", name: "Balance collected",                     actor: "customer" },
  { key: "closed",    phase: "close", name: "Closed",                                actor: "tr", hide: true },
];
function mintTok() {
  return [...crypto.getRandomValues(new Uint8Array(24))]
    .map((x) => x.toString(16).padStart(2, "0")).join("");
}
// deno-lint-ignore no-explicit-any
async function seedJob(sb: any, jobId: string) {
  const { count } = await sb.from("job_steps").select("id", { count: "exact", head: true }).eq("job_id", jobId);
  if (!count) {
    await sb.from("job_steps").insert(STEP_RAIL.map((st, i) => ({
      job_id: jobId, sort: (i + 1) * 10, phase: st.phase, key: st.key, name: st.name,
      actor: st.actor, visible_customer: !st.hide,
    })));
  }
  const { data: j } = await sb.from("jobs").select("verify_token, track_token").eq("id", jobId).single();
  const patch: Record<string, unknown> = {};
  if (!j?.verify_token) patch.verify_token = mintTok();
  if (!j?.track_token) patch.track_token = mintTok();
  if (Object.keys(patch).length) await sb.from("jobs").update(patch).eq("id", jobId);
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: CORS });
  try {
    const KEY = Deno.env.get("INTAKE_KEY");
    const SB_KEY = Deno.env.get("SB_SECRET_KEY") ?? Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
    if (!KEY || !SB_KEY) return out({ ok: false, error: "MISSING SECRET: INTAKE_KEY or SB_SECRET_KEY" }, 500);

    const sb = createClient(Deno.env.get("SUPABASE_URL")!, SB_KEY);
    const b = await req.json().catch(() => ({}));

    // ---------- PUBLIC: web enquiry (the Bifolded.com front door) ----------
    // No office key — this is the one action the public may call. Honeypot:
    // bots fill every field; a human never sees "website". Fake success so
    // bots move on; nothing is written.
    if (b.action === "enquiry") {
      if (String(b.website ?? "").trim() !== "") return out({ ok: true, label: "received" });
      const name = String(b.name ?? "").trim().replace(/\s+/g, " ").slice(0, 120);
      const email = String(b.email ?? "").trim().slice(0, 160);
      const phone = String(b.phone ?? "").trim().slice(0, 40);
      if (name.length < 2) return out({ ok: false, error: "Please give us a name." }, 400);
      if (!email && !phone) return out({ ok: false, error: "An email or a phone number — so we can reach you back." }, 400);

      const { data: dupes } = await sb.from("customers")
        .select("id, customer_no").ilike("name", name).limit(1);
      const repeat = (dupes?.length ?? 0) > 0;

      const { data: cno, error: mintErr } = await sb.rpc("mint_no", { p_kind: "customer" });
      if (mintErr) return out({ ok: false, error: "Something went wrong on our side — please call us." }, 500);

      const cfg = (typeof b.config === "object" && b.config) || null;
      // internal estimate from the interim practice model (M&T base) —
      // office eyes only, never returned to the caller
      let est: number | null = null;
      if (cfg && Number(cfg.w) > 0 && Number(cfg.h) > 0) {
        const sq = Number(cfg.w) * Number(cfg.h);
        const glazed = /glass/i.test(String(cfg.infill ?? ""));
        est = Math.round(8806 + 70.86 * sq + (glazed ? 2932 : 0));
      }
      const notes = [
        b.interest ? "Interested in: " + String(b.interest).slice(0, 200) : "",
        b.size ? "Rough opening: " + String(b.size).slice(0, 120) : "",
        (b.message ?? b.details) ? "In their words: " + String(b.message ?? b.details).slice(0, 2000) : "",
        cfg ? "Web configurator: " + cfg.w + "\u00d7" + cfg.h + " ft, " + cfg.panels + " panels, frame " +
          String(cfg.frame ?? "?") + ", infill " + String(cfg.infill ?? "?") + ", automation " + String(cfg.automation ?? "?") : "",
        est != null ? "Internal est \u2248 $" + est.toLocaleString("en-US") + " (interim M&T model \u2014 office only, not quoted)" : "",
        repeat ? "(possible repeat — a customer with this exact name already exists)" : "",
      ].filter(Boolean).join("\n");

      const { data: created, error } = await sb.from("customers").insert({
        customer_no: cno, name, display_name: name, status: "prospect",
        account_type: "unknown", email: email || null, phone: phone || null,
        site_address: String(b.city ?? b.site ?? "").trim().slice(0, 160) || null,
        source: "web-enquiry", notes: notes || null,
      }).select("id, customer_no, name").single();
      if (error) return out({ ok: false, error: "Something went wrong on our side — please call us." }, 500);

      await sb.from("events").insert({
        actor: "web", kind: "enquiry",
        body: { what: "web enquiry", customer_no: created.customer_no, customer_id: created.id, name: created.name, interest: b.interest ?? null, config: cfg, internal_est: est },
      });
      return out({ ok: true, label: "C-" + created.customer_no, name: created.name });
    }

    // ---------- PUBLIC: e-sign — the token IS the auth ----------
    if (b.action === "agree_view") {
      const tok = String(b.token ?? "").trim();
      if (tok.length < 20) return out({ ok: false, error: "bad link" }, 400);
      const { data: q, error } = await sb.from("quotes")
        .select("id, quote_no, status, value, agree_doc, signed_name, signed_at, customer_id")
        .eq("sign_token", tok).single();
      if (error || !q?.agree_doc) return out({ ok: false, error: "This link is not valid — call us and we will send a fresh one." }, 404);
      const { data: c } = await sb.from("customers").select("name, customer_no").eq("id", q.customer_id).single();
      return out({ ok: true, quote_no: q.quote_no, status: q.status, signed: !!q.signed_at,
        signed_name: q.signed_name, signed_at: q.signed_at, doc: q.agree_doc, customer: c });
    }

    if (b.action === "agree_sign") {
      const tok = String(b.token ?? "").trim();
      const name = String(b.name ?? "").trim().replace(/\s+/g, " ").slice(0, 120);
      if (tok.length < 20) return out({ ok: false, error: "bad link" }, 400);
      if (name.length < 2) return out({ ok: false, error: "Please sign with your full name." }, 400);
      if (b.consent !== true) return out({ ok: false, error: "Please tick the consent box — it is what makes your typed name a signature." }, 400);
      const { data: q, error: qe } = await sb.from("quotes")
        .select("id, quote_no, status, agree_doc, signed_at").eq("sign_token", tok).single();
      if (qe || !q?.agree_doc) return out({ ok: false, error: "This link is not valid — call us." }, 404);
      if (q.signed_at || q.status === "accepted") return out({ ok: false, error: "This agreement is already signed — welcome to the wait." }, 400);
      const d = (q.agree_doc as any)?.data ?? {};
      if (d.rush && !String(d.rushSignTR ?? "").trim()) {
        return out({ ok: false, error: "This agreement carries a rush guarantee without a TR sign-off — call us before signing." }, 400);
      }
      const enc = new TextEncoder().encode(JSON.stringify(q.agree_doc));
      const hash = [...new Uint8Array(await crypto.subtle.digest("SHA-256", enc))]
        .map((x) => x.toString(16).padStart(2, "0")).join("");
      const ip = (req.headers.get("x-forwarded-for") ?? "").split(",")[0].trim();
      const { error: ue } = await sb.from("quotes").update({
        signed_name: name, signed_at: new Date().toISOString(), signed_ip: ip || null, doc_hash: hash,
      }).eq("id", q.id);
      if (ue) return out({ ok: false, error: "Something went wrong on our side — call us." }, 500);
      const { data: jobId, error: ce } = await sb.rpc("close_quote", { p_quote_id: q.id });
      if (ce) return out({ ok: false, error: ce.message }, 500);
      if (d.rush) await sb.from("jobs").update({
        rush: true, rush_by: d.rushBy || null, rush_approved_by: d.rushSignTR || null }).eq("id", jobId);
      await seedJob(sb, jobId);
      await sb.from("events").insert({ actor: "customer", kind: "esign",
        body: { what: "agreement e-signed", quote_no: q.quote_no, signed_name: name, doc_hash: hash } });
      return out({ ok: true, signed: true, job_no: q.quote_no, name });
    }

    // ---------- PUBLIC: field verification (verify_token link) ----------
    if (b.action === "verify_view") {
      const tok = String(b.token ?? "").trim();
      if (tok.length < 20) return out({ ok: false, error: "bad link" }, 400);
      const { data: j, error: je } = await sb.from("jobs")
        .select("id, job_no, status:stage, customer_id").eq("verify_token", tok).single();
      if (je || !j) return out({ ok: false, error: "This link is not valid — call us and we will send a fresh one." }, 404);
      const [{ data: c }, { data: people }, { data: vs }, { data: ms }] = await Promise.all([
        sb.from("customers").select("name, site_address").eq("id", j.customer_id).single(),
        sb.from("customer_contacts").select("name, role").eq("customer_id", j.customer_id).eq("active", true).order("rank"),
        sb.from("verifications").select("opening, by_name, at").eq("job_id", j.id).order("at", { ascending: false }),
        sb.from("measurements").select("opening, axis, governing, at").eq("job_id", j.id),
      ]);
      return out({ ok: true, job_no: j.job_no, customer: c?.name ?? "", site: c?.site_address ?? "",
        people: people ?? [], returned: vs ?? [], measured: ms ?? [] });
    }

    if (b.action === "verify_submit") {
      const tok = String(b.token ?? "").trim();
      if (tok.length < 20) return out({ ok: false, error: "bad link" }, 400);

    // ---------- PDF rail: PNG pages -> PDF -> job-files/docs + registry ----------
    // deno-lint-ignore no-explicit-any
    async function buildPdfFromPages(pages: any[]): Promise<Uint8Array> {
      const { PDFDocument } = await import("https://esm.sh/pdf-lib@1.17.1");
      const doc = await PDFDocument.create();
      for (const p of pages) {
        const m = /^data:image\/png;base64,(.+)$/.exec(String(p));
        if (!m) throw new Error("pages must be PNG data URLs");
        const bytes = Uint8Array.from(atob(m[1]), (c) => c.charCodeAt(0));
        const img = await doc.embedPng(bytes);
        const pg = doc.addPage([612, 792]);
        const mgn = 24, aw = 612 - 2 * mgn, ah = 792 - 2 * mgn;
        const k = Math.min(aw / img.width, ah / img.height);
        const w = img.width * k, h = img.height * k;
        pg.drawImage(img, { x: (612 - w) / 2, y: 792 - mgn - h, width: w, height: h });
      }
      return await doc.save();
    }
    // deno-lint-ignore no-explicit-any
    async function fileDoc(o: any): Promise<any> {
      if (!Array.isArray(o.pages) || !o.pages.length || o.pages.length > 12) return { error: "1-12 PNG pages required" };
      let tot = 0; for (const p of o.pages) tot += String(p).length;
      if (tot > 24_000_000) return { error: "pages too large" };
      let bytes: Uint8Array;
      try { bytes = await buildPdfFromPages(o.pages); } catch (e) { return { error: "pdf: " + (e as Error).message }; }
      const hash = [...new Uint8Array(await crypto.subtle.digest("SHA-256", bytes as unknown as ArrayBuffer))].map((x) => x.toString(16).padStart(2, "0")).join("");
      const stamp = new Date().toISOString().replace(/[:.]/g, "-").slice(0, 19);
      const path = "docs/" + String(o.kind ?? "doc") + "/" + String(o.ref ?? "misc") + "/" + String(o.kind ?? "doc") + "-" + stamp + ".pdf";
      const { error: se } = await sb.storage.from("job-files").upload(path, bytes, { contentType: "application/pdf", upsert: true });
      if (se) return { error: se.message };
      const { data, error } = await sb.from("doc_files").insert({
        kind: String(o.kind ?? "doc"), title: String(o.title ?? "").slice(0, 200) || null,
        quote_id: o.quote_id ? String(o.quote_id) : null, concept_id: o.concept_id ? String(o.concept_id) : null,
        job_id: o.job_id ? String(o.job_id) : null,
        path, sha256: hash, pages: o.pages.length, made_by: String(o.by ?? "system"),
      }).select("*").single();
      if (error) return { error: error.message };
      await sb.from("events").insert({ actor: String(o.by ?? "system"), kind: "doc",
        body: { what: String(o.kind) + " PDF filed", path, pages: o.pages.length } });
      return { doc: data };
    }

    // ---------- ORDERING ENGINE — kit math ported verbatim from Shop Ops 3.49 ----------
    // Centor E3 base hardware per pivoted-jamb group of k doors (verified vs
    // Centor "E3 Product Line" Common Panel Layouts pp.16-17; k=6 extrapolated).
    const E3_BASE: Record<number, Record<string, number>> = {
      1: { ics: 0, end: 0, hs: 0, hhs: 0, db: 0, pass: 1 },
      2: { ics: 0, end: 1, hs: 1, hhs: 0, db: 2, pass: 0 },
      3: { ics: 1, end: 0, hs: 0, hhs: 1, db: 2, pass: 1 },
      4: { ics: 1, end: 1, hs: 0, hhs: 2, db: 4, pass: 0 },
      5: { ics: 2, end: 0, hs: 1, hhs: 1, db: 4, pass: 1 },
      6: { ics: 2, end: 1, hs: 1, hhs: 2, db: 6, pass: 0, est: 1 },
      7: { ics: 3, end: 0, hs: 2, hhs: 1, db: 6, pass: 1 },
      8: { ics: 3, end: 1, hs: 2, hhs: 2, db: 8, pass: 0 },
    };
    const E3_TALL = 88; // over 7'-4" (2250mm): wall pivot set + 4th hinge
    // deno-lint-ignore no-explicit-any
    function e3Config(dl: number, dr: number): any {
      const o = { code: (dl ? dl + "L" : "") + (dr ? dr + "R" : ""), ps: 0, wps: 0, ics: 0, lcs: 0, rcs: 0, hs: 0, hhs: 0, db: 0, pass: 0, est: false };
      for (const [k, side] of [[dl, "L"], [dr, "R"]] as [number, string][]) {
        if (!k) continue;
        const b2 = E3_BASE[k];
        if (!b2) { o.est = true; continue; }
        o.ps += 1; o.wps += 1; o.ics += b2.ics; o.hs += b2.hs; o.hhs += b2.hhs; o.db += b2.db; o.pass += b2.pass;
        if (b2.end) { if (side === "L") o.rcs += 1; else o.lcs += 1; }
        if (b2.est) o.est = true;
      }
      if (o.pass > 1) { o.db += 2 * (o.pass - 1); o.pass = 1; }
      return o;
    }
    // deno-lint-ignore no-explicit-any
    function kitLines(sys: string, n: number, c: any): { name: string; code?: string; qty: number; unit?: string; note?: string }[] {
      const L: { name: string; code?: string; qty: number; unit?: string; note?: string }[] = [];
      const tall = Number(c.leafH) > (sys === "e3" ? E3_TALL : 96);
      // WOODWORK (the paper 4-PANEL BI-FOLD PARTS LIST, modernized)

            rush_approved_by: (rushPatch.rush_approved_by as string) ?? null,

          body: { what: "agreement signed — quote closed", quote_no: q.quote_no,
                  rush: !!rushPatch.rush, rush_by: rushPatch.rush_by ?? null },
        });
        return out({ ok: true, closed: true, job_id: jobId, job_no: q.quote_no });
      }

      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("events").insert({
        actor: String(b.by ?? "office"), kind: "agree",
        body: { what: "agreement " + status, quote_no: q.quote_no },

      return out({ ok: true, status });

        .select("*").eq("book_id", b.book_id).order("sort");

        .select("id, book_id").eq("id", b.item_id).single();

      if (b.note !== undefined) patch.note = String(b.note).trim();

        sb.from("jobs").select("id", { count: "exact", head: true }).eq("customer_id", cid),

      if (error) return out({ ok: false, error: error.message }, 500);

      if (b.role !== undefined) patch.role = ["owner", "contractor", "architect", "designer", "builder", "other"].includes(b.role) ? b.role : "other";
      if (b.preferred_method !== undefined) patch.preferred_method = ["call", "text", "email"].includes(b.preferred_method) ? b.preferred_method : null;
      if (b.rank !== undefined) patch.rank = Math.max(1, Math.min(Number(b.rank) || 9, 9));
      if (b.active !== undefined) patch.active = !!b.active;

      for (const st of steps ?? []) {

      const [{ data: c }, { data: people }, { data: steps }, { data: ms }, { data: vs }, { data: duties }, { data: labor }, { data: photos }] = await Promise.all([
        sb.from("customers").select("id, customer_no, name, email, phone, site_address").eq("id", j.customer_id).single(),
        sb.from("customer_contacts").select("*").eq("customer_id", j.customer_id).order("rank"),
        sb.from("job_steps").select("*").eq("job_id", jid).order("sort"),
        sb.from("measurements").select("*").eq("job_id", jid).order("at", { ascending: false }),

        sb.from("labor_entries").select("*").eq("job_id", jid).order("at", { ascending: false }),

    // ---------- laser absorption: the calcs sheet is now server math ----------
    // deno-lint-ignore no-explicit-any
    async function laserFor(jobId: string): Promise<any> {
      if (!jobId) return null;
      const { data: ms } = await sb.from("measurements").select("axis, legs, governing, by_name, at")
        .eq("job_id", jobId).eq("kind", "laser").order("at", { ascending: false });
      if (!ms?.length) return null;
      // deno-lint-ignore no-explicit-any
      const w = (ms as any[]).find((m) => m.axis === "w"), h = (ms as any[]).find((m) => m.axis === "h");
      if (!w || !h) return null;
      const spread = (legs: number[]) => legs.length > 1 ? Math.max(...legs) - Math.min(...legs) : 0;
      return { max_w: Number(w.governing), max_h: Number(h.governing),
        w_legs: w.legs, h_legs: h.legs,
        spread_w: spread(w.legs ?? []), spread_h: spread(h.legs ?? []),
        out_of_square: spread(w.legs ?? []) > 0.375 || spread(h.legs ?? []) > 0.375,
        by_name: w.by_name, at: w.at };
    }
    // deno-lint-ignore no-explicit-any
    function laserBuild(lz: any, spec: any) {
      if (spec?.mount === "backset") {
        // REAR FLUSH: unit oversizes the opening — trim conceals header/jambs
        const ov = (spec?.overlap && typeof spec.overlap === "object") ? spec.overlap : {};
        const oside = Number(ov.side) > 0 ? Number(ov.side) : 2;
        const ohead = Number(ov.head) > 0 ? Number(ov.head) : 2;
        return { mount: "backset", build_w: Math.round((lz.max_w + 2 * oside) * 32) / 32,
          build_h: Math.round((lz.max_h + ohead) * 32) / 32, side: oside, head: ohead };
      }
      const g = (spec?.gaps && typeof spec.gaps === "object") ? spec.gaps : {};
      const side = Number(g.side) > 0 ? Number(g.side) : 0.5;
      const head = Number(g.head) > 0 ? Number(g.head) : 0.5;
      return { mount: "inset", build_w: Math.round((lz.max_w - 2 * side) * 32) / 32, build_h: Math.round((lz.max_h - head) * 32) / 32, side, head };
    }

    // ---------- M8/M10 track: Squizz's Weight Calc 1.6, concept-stage ----------
    // Cavity Sliders carriages M6 200 / M8 500 / M10 1100 lb (verified 08-2026);
    // carriage load = share x leaf (Drive Force Calc statics). Every constant is
    // tunable in catalog_options track_cfg/default — his numbers, not guesses.
    const TRACK_DEF = { m6: 200, m8: 500, m10: 1100, margin: 0.8, share: 0.5, hw_lb: 18,
      fam_psf: { tb: 4.0, carriage: 5.0, rs_paint: 5.0, rs_okoume: 4.5, rs_soft: 5.0, rs_hard: 6.5, glass: 5.5 },
      clad_psf: 3.8 };
    // deno-lint-ignore no-explicit-any
    async function trackCfg(): Promise<any> {
      const { data } = await sb.from("catalog_options").select("data")
        .eq("group_key", "track_cfg").eq("opt_id", "default").maybeSingle();
      // deno-lint-ignore no-explicit-any
      const d = ((data as any)?.data && typeof (data as any).data === "object") ? (data as any).data : {};
      return { ...TRACK_DEF, ...d, fam_psf: { ...TRACK_DEF.fam_psf, ...(d.fam_psf ?? {}) } };
    }
    // deno-lint-ignore no-explicit-any
    function trackFor(spec: any, cfg: any) {
      const doors = Math.max(1, Number(spec?.doors) || 1);
      const w = Number(spec?.width_in) || 0, h = Number(spec?.height_in) || 0;
      if (!w || !h) return null;
      const area = (w / doors) * h / 144;
      const fam = String(spec?.family ?? "");
      const F = cfg.fam_psf;
      const base = fam.startsWith("carriage") ? F.carriage
        : fam === "paintgrade" ? F.rs_paint : fam === "okoume" ? F.rs_okoume
        : fam === "softwood" ? F.rs_soft : fam === "hardwood" ? F.rs_hard
        : (fam === "glasspanel" || fam === "fullview") ? F.glass : F.tb;
      let clad = 0;
      const cw = Number((spec?.clad ?? {}).weight);
      if (cw > 0) clad = cw > 20 ? cw / area : cw; // a big number was lb per leaf
      else if (spec?.cladding === "tr" || spec?.cladding === "owner" || fam.startsWith("tbclad")) clad = Number(cfg.clad_psf) || 0;
      const leaf = Math.round((area * (base + clad) + Number(cfg.hw_lb)) * 10) / 10;
      const load = Math.round(leaf * Number(cfg.share) * 10) / 10;
      const mg = Number(cfg.margin) > 0 && Number(cfg.margin) <= 1 ? Number(cfg.margin) : 0.8;
      const auto = load <= Number(cfg.m8) * mg ? "m8" : load <= Number(cfg.m10) * mg ? "m10" : "over";
      const forced = spec?.track === "m8" || spec?.track === "m10";
      const track = forced ? spec.track : (auto === "over" ? "m10" : auto);
      const cap = track === "m8" ? Number(cfg.m8) : Number(cfg.m10);
      const ok = load <= cap * mg + 0.001;
      const clad_allow = Math.floor(cap * mg / Number(cfg.share) - (area * base + Number(cfg.hw_lb)));
      return { leaf_lb: leaf, carriage_lb: load, auto, track, forced, cap, margin: mg, ok,
        over_m10: auto === "over", clad_allow_lb: Math.max(0, clad_allow),
        area_sqft: Math.round(area * 10) / 10, base_psf: base, clad_psf: clad };
    }

          const rmin = Number(sp.reveal_min) > 0 ? Number(sp.reveal_min) : 0.25;
          if (Number(sp.width_in) < lz.max_w + 2 * rmin - 0.001 || Number(sp.height_in) < lz.max_h + rmin - 0.001) {
            return out({ ok: false, error: "Backset flush must OVERSIZE the opening — at least " + rmin + " in reveal per side at minimum lasers (opening max " + lz.max_w + " × " + lz.max_h + "). Adopt the backset build, or increase the size." }, 400);
          }
        } else if (lz && (Number(sp.width_in) > lz.max_w + 0.01 || Number(sp.height_in) > lz.max_h + 0.01)) {
          const ovr = sp.laser_override ?? {};
          const okOvr = ["site_change", "customer_certified"].includes(ovr.kind) && String(ovr.reason ?? "").trim().length >= 10;
          if (!okOvr) {
            return out({ ok: false, error: "Over the verified lasers (max " + lz.max_w + " × " + lz.max_h + "). Adopt the verified size, or record the override — site being modified, or customer certifies the number (with the reason)." }, 400);
          }
        }
      }
      const token = c.view_token ?? [...crypto.getRandomValues(new Uint8Array(24))]
        .map((x: number) => x.toString(16).padStart(2, "0")).join("");
      const { error } = await sb.from("concepts").update({
        view_token: token, status: "sent", sent_at: new Date().toISOString(), viewed_at: null,
      }).eq("id", c.id);
      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("events").insert({ actor: String(b.by ?? "office"), kind: "concept",
        body: { what: c.stage + " published", title: c.title, rev: c.rev } });
      return out({ ok: true, token });
    }

    if (b.action === "concept_del") {
      if (b.confirm !== true) return out({ ok: false, error: "confirm:true required" }, 400);
      const { data: c } = await sb.from("concepts").select("status, title").eq("id", String(b.id ?? "")).single();
      if (!c) return out({ ok: false, error: "not found" }, 404);
      if (c.status !== "draft") return out({ ok: false, error: "Only drafts delete — sent history is kept." }, 400);
      const { error } = await sb.from("concepts").delete().eq("id", String(b.id));
      if (error) return out({ ok: false, error: error.message }, 500);
      return out({ ok: true, deleted: true });
    }

    if (b.action === "pin_set") {
      const patch: Record<string, unknown> = {};
      if (b.status === "resolved") { patch.status = "resolved"; patch.resolved_by = String(b.by ?? "office");
        patch.resolved_at = new Date().toISOString(); patch.resolved_note = String(b.resolved_note ?? "").trim() || null; }
      else if (b.status === "open") { patch.status = "open"; patch.resolved_at = null; patch.resolved_by = null; }
      if (!Object.keys(patch).length) return out({ ok: false, error: "nothing to change" }, 400);
      const { data, error } = await sb.from("concept_pins").update(patch).eq("id", String(b.id ?? "")).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      return out({ ok: true, pin: data });
    }

    // ---------- PDF rail (office): make / list / link / send ----------
    if (b.action === "pdf_make") {
      const r = await fileDoc({ kind: String(b.kind ?? "doc"), ref: String(b.ref ?? "misc"),
        quote_id: b.quote_id ?? null, concept_id: b.concept_id ?? null, job_id: b.job_id ?? null,
        title: b.title, pages: b.pages, by: String(b.by ?? "office") });
      if (r.error) return out({ ok: false, error: r.error }, 400);
      const { data: su } = await sb.storage.from("job-files").createSignedUrl(r.doc.path, 604800);
      return out({ ok: true, doc: r.doc, url: su?.signedUrl ?? null });
    }

    if (b.action === "docs_list") {
      let q = sb.from("doc_files").select("*").order("created_at", { ascending: false }).limit(100);
      if (b.job_id) q = q.eq("job_id", String(b.job_id));
      if (b.concept_id) q = q.eq("concept_id", String(b.concept_id));
      if (b.quote_id) q = q.eq("quote_id", String(b.quote_id));
      const { data, error } = await q;
      if (error) return out({ ok: false, error: error.message }, 500);
      return out({ ok: true, rows: data ?? [] });
    }

    if (b.action === "doc_url") {
      const { data: d } = await sb.from("doc_files").select("path").eq("id", String(b.id ?? "")).single();
      if (!d) return out({ ok: false, error: "not found" }, 404);
      const { data: su, error } = await sb.storage.from("job-files").createSignedUrl(d.path, 604800);
      if (error) return out({ ok: false, error: error.message }, 500);
      return out({ ok: true, url: su.signedUrl });
    }

    if (b.action === "doc_send") {
      // sends AS the real person — their TR email (Squizz: "they send everything
      // thru their TR emails"). Needs RESEND_API_KEY + the domain verified;
      // until then it hands back the 7-day link to paste anywhere.
      const { data: d } = await sb.from("doc_files").select("*").eq("id", String(b.id ?? "")).single();
      if (!d) return out({ ok: false, error: "not found" }, 404);
      const to = String(b.to ?? "").trim();
      if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(to)) return out({ ok: false, error: "a valid to email is required" }, 400);
      const { data: su } = await sb.storage.from("job-files").createSignedUrl(d.path, 604800);
      const link = su?.signedUrl ?? "";
      const rk = Deno.env.get("RESEND_API_KEY");
      const fromName = String(b.from_name ?? "Tungsten Royce").slice(0, 80);
      const fromEmail = String(b.from_email ?? "").trim();
      if (!rk || !fromEmail) {
        return out({ ok: true, sent: false, link,
          note: !rk ? "Email not configured yet (RESEND_API_KEY missing) — copy the link." : "That sender has no email saved — copy the link, or set their email." });
      }
      const subject = String(b.subject ?? (d.title || "Your document from Tungsten Royce")).slice(0, 150);
      const msg = String(b.message ?? "").slice(0, 2000);
      const esc9 = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;");
      const html = "<p>" + esc9(msg).replace(/\n/g, "<br>") + "</p><p><a href=\"" + link + "\">Download " + esc9(d.title || "your document") + " (PDF)</a><br><span style=\"color:#777;font-size:12px\">Link valid 7 days.</span></p>";
      const resp = await fetch("https://api.resend.com/emails", { method: "POST",
        headers: { "content-type": "application/json", Authorization: "Bearer " + rk },
        body: JSON.stringify({ from: fromName + " <" + fromEmail + ">", to: [to], reply_to: fromEmail, subject, html }) });
      // deno-lint-ignore no-explicit-any
      const rj: any = await resp.json().catch(() => ({}));
      if (!resp.ok) return out({ ok: false, error: "send failed: " + (rj?.message ?? resp.status), link }, 502);
      await sb.from("doc_files").update({ sent_to: to, sent_at: new Date().toISOString(), sent_by: String(b.by ?? "office") }).eq("id", d.id);
      await sb.from("events").insert({ actor: String(b.by ?? "office"), kind: "doc", body: { what: "doc emailed", to, title: d.title } });
      return out({ ok: true, sent: true, id: rj?.id ?? null, link });
    }

    // ---------- TRAVELERS (office): spawn from fab_steps templates ----------
    if (b.action === "trav_spawn") {
      const tmpl = String(b.template ?? "").trim();
      const { data: opt } = await sb.from("catalog_options").select("name, data").eq("group_key", "fab_steps").eq("opt_id", tmpl).single();
      if (!opt) return out({ ok: false, error: "unknown template — pick one from Fab Steps" }, 400);
      // deno-lint-ignore no-explicit-any
      const steps: string[] = ((opt.data as any)?.steps ?? []).filter((s: unknown) => typeof s === "string");
      if (!steps.length) return out({ ok: false, error: "that template has no steps" }, 400);
      let jobNo: string | null = null;
      if (b.job_id) { const { data: j } = await sb.from("jobs").select("job_no").eq("id", String(b.job_id)).single(); jobNo = j?.job_no ?? null; }
      const { data: tv, error } = await sb.from("travelers").insert({
        job_id: b.job_id ? String(b.job_id) : null, job_no: jobNo, template: tmpl,
        name: String(b.name ?? "").trim().slice(0, 160) || (opt.name + (jobNo ? " — " + jobNo : "")),
        worker: String(b.worker ?? "").trim() || null, created_by: String(b.by ?? "office"),
      }).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("traveler_steps").insert(steps.map((s, i) => ({ traveler_id: tv.id, sort: i * 10, text: s.slice(0, 400) })));
      if (tv.worker) await sb.from("duties").insert({ job_id: tv.job_id, title: "[TRAVELER] " + tv.name, worker: tv.worker, status: "now", sort: 90, pushed_by: String(b.by ?? "office") });
      await sb.from("events").insert({ actor: String(b.by ?? "office"), kind: "traveler",
        body: { what: "traveler spawned", name: tv.name, steps: steps.length, worker: tv.worker } });
      return out({ ok: true, traveler: tv, steps: steps.length });
    }

    if (b.action === "travs_list") {
      let q = sb.from("travelers").select("*").order("created_at", { ascending: false }).limit(80);
      if (b.job_id) q = q.eq("job_id", String(b.job_id));
      if (b.open === true) q = q.eq("status", "open");
      const { data: travs, error } = await q;
      if (error) return out({ ok: false, error: error.message }, 500);
      const ids = (travs ?? []).map((t) => t.id);
      const { data: steps } = ids.length ? await sb.from("traveler_steps").select("*").in("traveler_id", ids).order("sort") : { data: [] };
      return out({ ok: true, travelers: travs ?? [], steps: steps ?? [] });
    }

    if (b.action === "trav_step_set") {
      const st = ["todo", "done", "na"].includes(b.status) ? b.status : "done";
      const { data, error } = await sb.from("traveler_steps").update({
        status: st, done_by: st === "todo" ? null : String(b.by ?? "office"),
        done_at: st === "todo" ? null : new Date().toISOString(),
        note: b.note !== undefined ? String(b.note).slice(0, 300) : undefined,
      }).eq("id", String(b.id ?? "")).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      const { data: rest } = await sb.from("traveler_steps").select("status").eq("traveler_id", data.traveler_id);
      const allDone = (rest ?? []).every((s) => ["done", "na"].includes(s.status));
      await sb.from("travelers").update({ status: allDone ? "done" : "open", done_at: allDone ? new Date().toISOString() : null }).eq("id", data.traveler_id);
      return out({ ok: true, step: data, traveler_done: allDone });
    }

    if (b.action === "trav_del") {
      if (b.confirm !== true) return out({ ok: false, error: "confirm:true required" }, 400);
      const { error } = await sb.from("travelers").delete().eq("id", String(b.id ?? ""));
      if (error) return out({ ok: false, error: error.message }, 500);
      return out({ ok: true, deleted: true });
    }

    // ---------- the flag rail (office ⇄ office; sales later) ----------
    if (b.action === "flag_add") {
      const from = String(b.by ?? "").trim().slice(0, 60);
      const to = String(b.to_name ?? "").trim().slice(0, 60);
      const q = String(b.question ?? "").trim().slice(0, 1000);
      if (from.length < 2 || to.length < 2 || q.length < 3) {
        return out({ ok: false, error: "who it is from, who it is for, and the question" }, 400);
      }
      const { data, error } = await sb.from("office_flags").insert({
        from_name: from, to_name: to, question: q,
        job_id: String(b.job_id ?? "") || null,
        concept_id: String(b.concept_id ?? "") || null,
        file_link: String(b.file_link ?? "").trim().slice(0, 400) || null,
      }).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("events").insert({ actor: from, kind: "flag",
        body: { what: "flag raised", to, question: q.slice(0, 120) } });
      return out({ ok: true, flag: data });
    }

    if (b.action === "flag_set") {
      const status = ["received", "working", "resolved"].includes(b.status) ? b.status : null;
      if (!status) return out({ ok: false, error: "status must be received | working | resolved" }, 400);
      const patch: Record<string, unknown> = { status };
      const now = new Date().toISOString();
      if (status === "received") patch.received_at = now;
      if (status === "working") patch.working_at = now;
      if (status === "resolved") {
        const rn = String(b.resolution ?? "").trim().slice(0, 1000);
        if (rn.length < 3) return out({ ok: false, error: "A resolution note is required — what was actually done." }, 400);
        patch.resolved_at = now; patch.resolution = rn;
      }
      const { data, error } = await sb.from("office_flags").update(patch).eq("id", String(b.id ?? "")).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("events").insert({ actor: String(b.by ?? "office"), kind: "flag",
        body: { what: "flag " + status, question: String(data.question).slice(0, 120), resolution: data.resolution } });
      return out({ ok: true, flag: data });
    }

    if (b.action === "flag_verdict") {
      const verdict = ["done_all", "partial", "not_done"].includes(b.verdict) ? b.verdict : null;
      if (!verdict) return out({ ok: false, error: "verdict must be done_all | partial | not_done" }, 400);
      const { data, error } = await sb.from("office_flags").update({ verdict }).eq("id", String(b.id ?? "")).select("*").single();
      if (error) return out({ ok: false, error: error.message }, 500);
      await sb.from("events").insert({ actor: String(b.by ?? "office"), kind: "flag",
        body: { what: "verdict " + verdict, question: String(data.question).slice(0, 120) } });
      return out({ ok: true, flag: data });
    }

    if (b.action === "flags_list") {
      const { data: rows } = await sb.from("office_flags").select("*")
        .gte("created_at", new Date(Date.now() - 90 * 86400000).toISOString())
        .order("created_at", { ascending: false }).limit(200);
      const jids = [...new Set((rows ?? []).map((f: { job_id: string | null }) => f.job_id).filter(Boolean))] as string[];
      const jobNames: Record<string, string> = {};
      if (jids.length) {
        const { data: js } = await sb.from("jobs").select("id, job_no").in("id", jids);
        for (const j of js ?? []) jobNames[(j as { id: string }).id] = (j as { job_no: string }).job_no;
      }
      return out({ ok: true, rows: rows ?? [], job_names: jobNames });
    }

    return out({ ok: false, error: "unknown action" }, 400);
  } catch (e) {
    return out({ ok: false, error: String(e?.message ?? e) }, 500);
  }
});
